# Código Orientado a Objetos - AquaBot Dynamics
class FlotadorArquimedes:
    def __init__(self, botellas=2):
        self.botellas = botellas
        self.en_agua = False

    def verificar_flotabilidad(self, esta_en_agua):
        self.en_agua = esta_en_agua
        return "Flotabilidad Positiva (Arquímedes Activo)" if self.en_agua else "En Superficie Seca"

class MotorPropulsion:
    def __init__(self, lado):
        self.lado = lado
        self.velocidad = 0

    def encender(self, potencia=100):
        self.velocidad = potencia
        print(f"Motor {self.lado}: Girando a {self.velocidad}% potencia.")

    def detener(self):
        self.velocidad = 0
        print(f"Motor {self.lado}: Detenido.")

class BarcoRobot:
    def __init__(self):
        self.flotador = FlotadorArquimedes()
        self.motor_izq = MotorPropulsion("Izquierdo")
        self.motor_der = MotorPropulsion("Derecho")
        self.bluetooth_conectado = False

    def conectar_bluetooth(self):
        self.bluetooth_conectado = True
        print("[HC-05] Conexión Bluetooth establecida con éxito.")

    def navegar_adelante(self):
        if self.bluetooth_conectado:
            self.motor_izq.encender(100)
            self.motor_der.encender(100)

    def detener_navegacion(self):
        self.motor_izq.detener()
        self.motor_der.detener()
